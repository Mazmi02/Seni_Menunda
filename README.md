            let questions = [];
            function generateQuestions() {
                rawVocabularyList.sort((a, b) => a.en.localeCompare(b.en));
                // Ambil semua daftar jawaban unik
                const allAnswers = [...new Set(rawVocabularyList.map(item => item.id))];
                questions = rawVocabularyList.map(item => {
                    // Ambil 3 pilihan salah secara acak tanpa loop while
                    const distractors = allAnswers
                    .filter(a => a !== item.id)
                    .sort(() => Math.random() - 0.5)
                    .slice(0, 3);
                    return {
                        question: item.en,
                        answers: [
                            { text: item.id, correct: true },
                            ...distractors.map(d => ({ text: d, correct: false }))
                        ]
                    };
                });
            }
            generateQuestions();
            function saveProgress() {
                if (!questionContainerElement.classList.contains('hide') && orderedQuestions && currentQuestionIndex < orderedQuestions.length) {
                    localforage.setItem('quizProgress_' + topicId, JSON.stringify({ currentQuestionIndex, score, orderedQuestions }));
                }
            }
            function loadProgress(callback) {
                localforage.getItem('quizProgress_' + topicId).then((data) => {
                    try { callback(data ? JSON.parse(data) : null); } catch (e) { callback(null); }
                }).catch(() => callback(null));
            }
            window.addEventListener('load', () => {
                loadProgress((data) => {
                    if (data) continueButton.classList.remove('hide');
                });
            });
            startButton.onclick = () => startGame(false);
            continueButton.onclick = () => startGame(true);
            function startGame(isContinuing) {
                clearTimeout(questionTimeout);
                loadProgress((savedData) => {
                    if (isContinuing && savedData) {
                        ({ orderedQuestions, currentQuestionIndex, score } = savedData);
                        currentZoomLevel = savedZoom;
                        zoomValueDisplay.innerText = currentZoomLevel;
                        applyZoom();
                    } else {
                        localforage.removeItem('quizProgress_' + topicId);
                        orderedQuestions = [...questions];
                        currentQuestionIndex = 0;
                        score = 0;
                        toggleAutoClickInput.checked = false;
                        zoomValueDisplay.innerText = currentZoomLevel;
                        timerValueDisplay.innerText = autoNextDelay;
                        nextQuestionButton.classList.toggle('hide', !toggleNextBtnInput.checked);
                        saveMenuSettings();
                        applyZoom();
                    }
                    initialControls.classList.add('hide');
                    questionContainerElement.classList.remove('hide');
                    questionCounterElement.classList.remove('hide');
                    completionMessageElement.classList.add('hide');
                    setNextQuestion();
                });
            }
            function setNextQuestion() {
                resetState();
                if (orderedQuestions && currentQuestionIndex < orderedQuestions.length) {
                    updateCounterDisplay(); showQuestion(orderedQuestions[currentQuestionIndex]);
                    saveProgress();
                } else if (orderedQuestions) { showResults(); }
                updateSkipButtonStates();
            }
            function showQuestion(q) {
                questionElement.innerText = q.question;
                answerButtonsElement.innerHTML = '';
                [...q.answers].sort(() => Math.random() - 0.5).forEach(a => {
                    const b = document.createElement('button');
                    b.innerText = a.text; b.classList.add('btn');
                    if (a.correct) b.dataset.correct = "true";
                    b.addEventListener('click', selectAnswer);
                    answerButtonsElement.appendChild(b);
                });
                if (toggleAutoClickInput.checked) {
                    autoClickTimeout = setTimeout(() => {
                        const c = Array.from(answerButtonsElement.children).find(b => b.dataset.correct === 'true');
                        if (c) selectAnswer({ target: c });
                    }, 2500);
                }
            }
            function resetState() {
                clearTimeout(questionTimeout); clearTimeout(autoClickTimeout);
                while (answerButtonsElement.firstChild) { answerButtonsElement.removeChild(answerButtonsElement.firstChild); }
            }
            function selectAnswer(e) {
                const btn = e.target;
                if (btn.dataset.correct === 'true') score++;
                Array.from(answerButtonsElement.children).forEach(b => {
                    b.classList.add(b.dataset.correct === 'true' ? 'correct' : 'wrong');
                    b.disabled = true;
                });
                saveProgress();
                questionTimeout = setTimeout(() => {
                    if (currentQuestionIndex < orderedQuestions.length - 1) { currentQuestionIndex++; setNextQuestion(); }
                    else showResults();
                }, autoNextDelay * 1000);
            }
            function showResults() {
                resetState();
                questionContainerElement.classList.add('hide');
                questionCounterElement.classList.add('hide');
                completionMessageElement.classList.remove('hide');
                initialControls.classList.remove('hide');
                startButton.innerText = 'Mulai';
                localforage.removeItem('quizProgress_' + topicId);
                continueButton.classList.add('hide');
            }
